┌─────────────────────────────────────────────────────────────────────┐
│ 🚀 每次启动必做（加载环境）                                         │
└─────────────────────────────────────────────────────────────────────┘
  cd ~/sztu && source setup.sh

┌─────────────────────────────────────────────────────────────────────┐
│ 📦 创建新功能包                                                      │
└─────────────────────────────────────────────────────────────────────┘
  # Python 包
  cd ~/sztu/src
  ros2 pkg create --build-type ament_python sztu_包名 \
      --dependencies rclpy sensor_msgs

  # C++ 包
  ros2 pkg create --build-type ament_cmake sztu_包名 \
      --dependencies rclcpp aimdk_msgs

┌─────────────────────────────────────────────────────────────────────┐
│ 🔨 编译                                                              │
└─────────────────────────────────────────────────────────────────────┘
  cd ~/sztu
  ./build_all.sh                            # 快速编译（推荐）
  colcon build --symlink-install            # 开发模式
  colcon build --packages-select 包名       # 只编译指定包
  
  编译后记得重新加载: source setup.sh

┌─────────────────────────────────────────────────────────────────────┐
│ ▶️  运行节点                                                          │
└─────────────────────────────────────────────────────────────────────┘
  source ~/sztu/setup.sh                    # 先加载环境
  ros2 run 包名 节点名                       # 运行节点
  ros2 launch 包名 launch文件.py            # 运行 launch

┌─────────────────────────────────────────────────────────────────────┐
│ 🔍 调试与查看                                                        │
└─────────────────────────────────────────────────────────────────────┘
  ros2 topic list                           # 所有 topic
  ros2 topic echo /topic_name               # 监听 topic
  ros2 node list                            # 所有节点
  ros2 node info /node_name                 # 节点详情
  ros2 service list                         # 所有服务
  ros2 pkg list | grep sztu                 # 查看 SZTU 的包

┌─────────────────────────────────────────────────────────────────────┐
│ 🧹 清理                                                              │
└─────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────┐
│ 🆘 常见问题                                                          │
└─────────────────────────────────────────────────────────────────────┘
  问题: 找不到 aimdk_msgs
  解决: source ~/sztu/setup.sh

  问题: Python 改动不生效
  解决: colcon build --symlink-install

  问题: 编译失败
  解决: rm -rf build/ install/ log/ && colcon build
